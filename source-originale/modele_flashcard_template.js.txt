function speak(text) {
  const src = VOICE_PACK[text];
  if (!src) {
    if ('speechSynthesis' in window) {
      try {
        speechSynthesis.cancel();
        const utterance = new SpeechSynthesisUtterance(text);
        utterance.lang = 'zh-CN';
        utterance.rate = 0.85;
        speechSynthesis.speak(utterance);
      } catch (e) {}
    }
    return;
  }
  try {
    audioEl.pause();
    audioEl.currentTime = 0;
    audioEl.src = src;
    audioEl.play().catch(() => {});
  } catch (e) {}
}

function t(k) {
  const d = I18N[preferredLang] || I18N.en;
  return (d[k] !== undefined ? d[k] : I18N.en[k]);
}
function tn(mode) { return t('modes')[mode]; }
function isRTL() { return preferredLang === 'ar'; }
function escapeHtml(s) {
  return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c]));
}
function norm(s) {
  // 用 \u2019 写 curly apostrophe，- 放末尾：避免老浏览器/Edge 对字符类解析差异
  let v = String(s).toLowerCase().replace(/[\s'\u2019·,.;:!?，。；：！？-]/g, '')
    .replace(/v/g, 'ü').replace(/u:/g, 'ü')
    .replace(/[āáǎà]/g, 'a').replace(/[ēéěè]/g, 'e').replace(/[īíǐì]/g, 'i')
    .replace(/[ōóǒò]/g, 'o').replace(/[ūúǔù]/g, 'u').replace(/[ǖǘǚǜ]/g, 'ü');
  v = v.replace(/^(j|q|x|y)ü/, '$1u');
  return v;
}
function fmt(n) { return String(Math.round(n * 10) / 10); }

// 按长度自动缩字号
function sizeClass(text, back) {
  const n = String(text).length;
  const p = back ? 'b' : '';
  if (n <= 9) return p + 'w1';
  if (n <= 15) return p + 'w2';
  if (n <= 24) return p + 'w3';
  if (n <= 40) return p + 'w4';
  return p + 'w5';
}
function blankDensity(n) { return n >= 10 ? 'xdense' : (n >= 7 ? 'dense' : ''); }

const WEIGHT_LABEL = '×';

/* ================= 轮次 ================= */
function startRound(mode) {
  curMode = mode;
  deck = shuffle(CARDS.slice());
  deck.forEach(c => { c._skip = 0; c._peeked = false; });
  roundTotal = deck.length;
  doneCount = 0;
  roundScore = 0;
  resetCard();
  updateHUD();
  render();
}

function shuffle(arr) {
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [arr[i], arr[j]] = [arr[j], arr[i]];
  }
  return arr;
}

function updateHUD() {
  doneEl.textContent = doneCount;
  totalEl.textContent = roundTotal;
  scoreLineEl.textContent = t('thisRound').replace('{s}', fmt(roundScore)) + ' · ' + t('totalAll').replace('{s}', fmt(totalScore));
}

function resetCard() {
  cardEl.style.transition = 'none';
  cardEl.classList.remove('flipped');
  void cardEl.offsetWidth;
  cardEl.style.transition = '';
}

function render() {
  if (!deck.length) { endRound(); return; }
  const c = deck[0];
  judged = false;
  peeked = !!c._peeked;        // 偷看标记本轮不清除（防止偷看后跳走再白拿分）
  const skipVisible = c._skip < 1;

  let f = '';
  if (curMode === 'translation') {
    const en = c.tr.en || '';
    const pref = preferredLang !== 'en' ? (c.tr[preferredLang] || '') : '';
    f += '<div class="content">';
    if (c.emoji) f += '<div class="emoji">' + c.emoji + '</div>';
    f += '<div class="word ' + sizeClass(en) + '">' + escapeHtml(en) + '</div>';
    if (pref) f += '<div class="sub">' + escapeHtml(pref) + '</div>';
    f += '</div>';
    f += '<div class="controls">' + rateRowHTML(skipVisible) + '</div>';

  } else if (curMode === 'listening') {
    f += '<div class="content">';
    f += '<div class="listen-icon" id="listenIcon">🔊</div>';
    f += '</div>';
    f += '<div class="controls">' + rateRowHTML(skipVisible) + '</div>';

  } else if (curMode === 'pinyin') {
    const en = c.tr.en || '';
    const pref = preferredLang !== 'en' ? (c.tr[preferredLang] || '') : '';
    f += '<div class="content">';
    if (c.emoji) f += '<div class="emoji">' + c.emoji + '</div>';
    f += '<div class="word ' + sizeClass(en) + '">' + escapeHtml(en) + '</div>';
    if (pref) f += '<div class="sub">' + escapeHtml(pref) + '</div>';
    f += blanksHTML(c);
    f += '<div class="tip-line" id="tipLine"></div>';
    f += '</div>';
    f += '<div class="controls">' + okRowHTML(skipVisible) + '</div>';

  } else { // hanzi
    f += '<div class="content">';
    if (c.emoji) f += '<div class="emoji">' + c.emoji + '</div>';
    f += '<div class="word ' + sizeClass(c.zh) + '">' + escapeHtml(c.zh) + '</div>';
    f += blanksHTML(c);
    f += '<div class="tip-line" id="tipLine"></div>';
    f += '</div>';
    f += '<div class="controls">' + okRowHTML(skipVisible) + '</div>';
  }

  frontEl.innerHTML = f;
  frontEl.setAttribute('dir', isRTL() ? 'rtl' : 'ltr');
  backEl.innerHTML = peekBackHTML(c);          // 预渲染答案，供偷看
  backEl.setAttribute('dir', isRTL() ? 'rtl' : 'ltr');
  bindFront(c);
  bindSpeak(c);
}

function rateRowHTML(skipVisible) {
  return '<div class="btn-row">' +
    '<button class="rate-btn red" data-v="0">' + escapeHtml(t('dontKnow')) + '</button>' +
    '<button class="rate-btn yellow" data-v="0">' + escapeHtml(t('unsure')) + '</button>' +
    '<button class="rate-btn green" data-v="1">' + escapeHtml(t('know')) + '</button>' +
    '</div>' +
    (skipVisible ? '<button class="skip-link" id="skipBtn">' + escapeHtml(t('skip')) + '</button>' : '');
}

function okRowHTML(skipVisible) {
  return '<div class="btn-row"><button class="ok-btn" id="okBtn">' + escapeHtml(t('ok')) + '</button></div>' +
    (skipVisible ? '<button class="skip-link" id="skipBtn">' + escapeHtml(t('skip')) + '</button>' : '');
}

function blanksHTML(c) {
  const den = blankDensity(c.syl.length);
  let h = '<div class="blanks ' + den + '">';
  c.syl.forEach((s, i) => {
    h += '<input class="pinyin-input" data-i="' + i + '" inputmode="latin" autocapitalize="off" autocorrect="off" spellcheck="false" maxlength="10">';
  });
  h += '</div>';
  return h;
}

function bindFront(c) {
  inputs = Array.from(frontEl.querySelectorAll('.pinyin-input'));
  inputs.forEach(inp => {
    inp.addEventListener('click', e => e.stopPropagation());
    inp.addEventListener('input', () => {
      const tip = document.getElementById('tipLine');
      if (tip) tip.textContent = '';
    });
    inp.addEventListener('keydown', e => {
      if (e.key === 'Enter') { e.preventDefault(); e.stopPropagation(); submitInput(c); }
    });
  });
  frontEl.querySelectorAll('.rate-btn').forEach(b => {
    b.addEventListener('click', e => {
      e.stopPropagation();
      submitRaw(parseFloat(b.getAttribute('data-v')), null, null);
    });
  });
  const ok = document.getElementById('okBtn');
  if (ok) ok.addEventListener('click', e => { e.stopPropagation(); submitInput(c); });
  const skip = document.getElementById('skipBtn');
  if (skip) skip.addEventListener('click', e => { e.stopPropagation(); doSkip(); });
  const li = document.getElementById('listenIcon');
  if (li) li.addEventListener('click', e => { e.stopPropagation(); speak(c.zh); });
}

/* ================= 提交与判定 ================= */
function submitInput(c) {
  if (judged) return;
  const empties = inputs.filter(i => i.value.trim() === '');
  if (empties.length) {
    const tip = document.getElementById('tipLine');
    if (tip) tip.textContent = t('fillAll');
    empties[0].focus();
    return;
  }
  const results = inputs.map((inp, i) => norm(inp.value) === norm(c.syl[i]));
  const raw = results.filter(Boolean).length / results.length;
  submitRaw(raw, results, c);
}

function submitRaw(raw, results, c) {
  if (judged) return;
  judged = true;
  const card = deck[0];

  // 容错：看过背面后作答
  let scored = raw;
  let note = '';
  if (peeked) {
    if (toleranceLeft > 0) {
      toleranceLeft--;
      note = t('peekUsed').replace('{n}', toleranceLeft);
    } else {
      scored = 0;
      note = t('peekZero');
    }
  }

  const gained = scored * WEIGHTS[curMode];
  roundScore += gained;
  totalScore += gained;
  doneCount++;

  // 标记输入框
  if (inputs.length) {
    inputs.forEach((inp, i) => {
      inp.classList.remove('bad', 'miss');
      inp.classList.add(results && results[i] ? 'ok' : 'bad');
      inp.readOnly = true;
    });
  }
  updateHUD();
  backEl.innerHTML = scoredBackHTML(card, gained, note, results);
  backEl.setAttribute('dir', isRTL() ? 'rtl' : 'ltr');
  bindBack(card);
  setTimeout(() => { cardEl.classList.add('flipped'); }, 380);
}

function bindSpeak(c) {
  const sb = document.getElementById('speakBtn');
  if (sb) sb.addEventListener('click', e => { e.stopPropagation(); speak(c.zh); });
}

function answerBodyHTML(c) {
  let h = '';
  if (curMode === 'pinyin' || curMode === 'translation') {
    h += '<div class="label">' + escapeHtml(t('zhLabel')) + '</div>';
    h += '<div class="meaning ' + sizeClass(c.zh, true) + '">' + escapeHtml(c.zh) + '</div>';
    h += '<div class="pinyin">' + escapeHtml(c.pinyin) + '</div>';
  } else if (curMode === 'listening') {
    h += '<div class="meaning ' + sizeClass(c.zh, true) + '">' + escapeHtml(c.zh) + '</div>';
    h += '<div class="pinyin">' + escapeHtml(c.pinyin) + '</div>';
    h += '<div class="explain">' + escapeHtml(c.tr.en || '') + '</div>';
    if (preferredLang !== 'en' && c.tr[preferredLang]) h += '<div class="explain">' + escapeHtml(c.tr[preferredLang]) + '</div>';
  } else { // hanzi
    h += '<div class="pinyin" style="font-size:22px;margin-top:0;">' + escapeHtml(c.pinyin) + '</div>';
    const ex = c.tr[preferredLang] || c.tr.en || '';
    if (ex) h += '<div class="explain">' + escapeHtml(ex) + '</div>';
  }
  return h;
}

// 偷看时看到的背面：只有答案，没有得分和「下一张」

// 作答后的背面：答案 + 本题得分 + 容错提示 + 下一张
function scoredBackHTML(c, gained, note, results) {
  let h = '<div class="content">' + answerBodyHTML(c);
  if (results) {
    h += '<div class="result-line">' + t('correct').replace('{c}', results.filter(Boolean).length).replace('{t}', results.length) + '</div>';
  }
  h += '<div class="result-line" style="font-weight:700;color:#8ef0c0;">+' + fmt(gained) + '</div>';
  h += '<div class="note-line">' + escapeHtml(note) + '</div>';
  h += speakBtnHTML(c);
  h += '</div>';
  h += '<div class="controls"><button class="next-btn" id="nextBtn">' + escapeHtml(t('next')) + '</button></div>';
  return h;
}

function bindBack(c) {
  const sb = document.getElementById('speakBtn');
  if (sb) sb.addEventListener('click', e => { e.stopPropagation(); speak(c.zh); });
  const nb = document.getElementById('nextBtn');
  if (nb) nb.addEventListener('click', e => { e.stopPropagation(); advance(); });
}

/* ================= 跳过 / 前进 ================= */
function doSkip() {
  if (transitioning || judged || !deck.length) return;
  const c = deck[0];
  if (c._skip >= 1) return;       // 每卡每轮最多跳 1 次
  c._skip++;
  deck.shift();
  deck.push(c);                   // 移到队尾，后面还会出现，但不给回看
  slideNext();
}

function advance() {
  if (transitioning || !deck.length) return;
  deck.shift();                    // 已作答，移出本轮
  if (!deck.length) { updateHUD(); endRound(); return; }
  slideNext();
}

function slideNext() {
  transitioning = true;
  resetCard();
  wrapperEl.style.transform = 'translateX(-120%)';
  setTimeout(() => {
    updateHUD();
    render();
    wrapperEl.style.transition = 'none';
    wrapperEl.style.transform = 'translateX(120%)';
    void wrapperEl.offsetWidth;
    wrapperEl.style.transition = '';
    wrapperEl.style.transform = 'translateX(0)';
    setTimeout(() => { transitioning = false; }, 280);
  }, 250);
}

function flip() {
  if (transitioning || judged) return;
  // 一旦翻到背面看过答案，本张卡本轮内就算「已偷看」，跳走再回来也不撤销
  if (!cardEl.classList.contains('flipped')) {
    peeked = true;
    if (deck[0]) deck[0]._peeked = true;
  }
  cardEl.classList.toggle('flipped');
}
document.getElementById('card').addEventListener('click', flip);
